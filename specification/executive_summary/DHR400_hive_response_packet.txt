# [A] NAMING CONVENTION:
# -------------------------------------------------------- |

  RES_TYPE : type of response
  RES_CODE : response return code
  RES_PRID : promise id
  RES_MPAR : multi-part response id
  RES_DATA : response data



# [B] RESPONSE PACKET REQUIRED STRUCTURE:
# -------------------------------------------------------- |

               + ---------------------------------------------------- +
               |        HEADER       |            PAYLOAD             |
               | ------------------- + ------------------------------ |
               |    1     |    2     |    3     |    4     |    5     |
  + --------- ++ -------- + -------- + -------- + -------- + -------- ++ ------------- +
  | COM_TYPE  || RES_TYPE | RES_CODE | RES_PRID | RES_MPAR | RES_DATA || response size |
  | --------- || -------- + -------- + -------- + -------- + -------- || ------------- |
  |      auth || 1        | <1, 4>   |          |          |    x     ||       3       |
  | --------- || -------- + -------- + -------- + -------- + -------- || ------------- |
  |   execute || 2        | <1, 4>   |          |          |    x     ||       3       |
  | --------- || -------- + -------- + -------- + -------- + -------- || ------------- |
  |   promise || 3        | <1, 4>   |    x     |          |    x     ||       4       |
  | --------- || -------- + -------- + -------- + -------- + -------- || ------------- |
  |    future || 4        | <0, 4>   |    x     |    x     |    x     ||       5       |
  + --------- ++ ---------------------------------------------------- ++ ------------- +
                                (max packet size : 10 x 1'024 = 10KB)


  + ----------------------------------------------------------------------------------------------------------------------------------------------------------- +

  + ----------- + ------------- + ----------------------------- + ----------------------------- + ----------------------------- + ----------------------------- +

  |             |               |                               |                               |                               |                               |
  + ----------- + ------------- + ----------------------------- + ----------------------------- + ----------------------------- + ----------------------------- +
  | execute     | 3             | [2,1,"DATA"]                  | [2,2,"DATA"]                  | [2,3,"err_RESSIZE"]           | [2,4,"err_REVOKED"]           |
  |             |               | [2,1,""]                      |                               |                               |                               |
  + ----------- + ------------- + ----------------------------- + ----------------------------- + ----------------------------- + ----------------------------- +
  | promise     | 4             | [3,1,PRID,"DATA"]             | [3,2,PRID,"DATA"]             | [3,3,PRID,"err_RESSIZE"]      | [3,4,PRID,"err_REVOKED"]      |

# [B1/1] HEADER: RES_TYPE:
# -------------------------------------------------------- |

  Related specification:
  ----------------------
  DHR200_hive_request_packet.txt

  + type of response
  + MUST match related request COM_TYPE
  + std::size_t

  + --------------------- +
  | COM_TYPE    | value   |
  + ----------- + ------- +
  | auth        |   1     |
  | ----------- + ------- |
  | execute     |   2     |
  | ----------- + ------- |
  | promise     |   3     |
  | ----------- + ------- |
  | future      |   4     |
  + --------------------- +



# [B1/1] HEADER: RES_CODE:
# -------------------------------------------------------- |

  + response return code
  + std::uint8_t

  + ------------------------------------------------------------------------------------------------------------- +
  | CODE    | meaning                | is static response     | description                                       |
  + ------- + ---------------------- + ---------------------- + ------------------------------------------------- +
  | 0       | awaiting response      | YES                    | COM_EXID execution result is enqueued -           |
  |         | COM_TYPE future only   |                        | result is not available yet. Try again later.     |
  + ------- + ---------------------- + ---------------------- + ------------------------------------------------- +
  | 1       | COM_EXID success       | NO                     | COM_EXID execution finished with a success.       |
  |         |                        | may have runtime msg   |                                                   |
  + ------- + ---------------------- + ---------------------- + ------------------------------------------------- +
  | 2       | COM_EXID failure       | NO                     | COM_EXID execution finished with an error.        |
  |         |                        | has runtime err msg    |                                                   |
  + ------- + ---------------------- + ---------------------- + ------------------------------------------------- +
  | 3       | response failure       | YES                    | A response failure may occur if any `COM_TYPE`    |
  |         |                        |                        | (specifically `auth`, `execute`, or `promise`)    |
  |         |                        |                        | attempts to return a buffer exceeding the         |
  |         |                        |                        | permitted size limit.                             |
  |         |                        | "err_RESSIZE"          |                                                   |
  |         |                        |                        | While this scenario is highly unlikely,           |
  |         |                        |                        | should it occur, the server owner must redefine   |
  |         |                        |                        | the request — for example, by switching the       |
  |         |                        |                        | `COM_TYPE` to an asynchronous communication       |
  |         |                        |                        | channel using a `promise` and `future`,           |
  |         |                        |                        | which supports multi-part responses by default.   |
  + ------- + ---------------------- + ---------------------- + ------------------------------------------------- +
  | 4       | system (FATAL) failure | YES                    | Access to DHR hive system has been revoked.       |
  |         |                        |                        | The server owner MUST process following steps:    |
  |         |                        | "err_REVOKED"          | [1] STOP server                                   |
  |         |                        |                        | [2] read logs                                     |
  |         |                        |                        | [3] fix the issue (fix error; ban skid; ...)      |
  |         |                        |                        | [4] START server                                  |
  + ------------------------------------------------------------------------------------------------------------- +



# [B2/1] PAYLOAD: RES_PRID:
# -------------------------------------------------------- |

  + promise id
  + std::size_t

  + ------------------------------------------------------ +
  | PRID     | description                                 |
  + -------- + ------------------------------------------- +
  |   0      | invalid promise ID                          |
  + -------- + ------------------------------------------- +
  | > 0      | valid promise ID                            |
  + ------------------------------------------------------ +



# [B2/2] PAYLOAD: RES_MPAR:
# -------------------------------------------------------- |

  + multi-part response id
  + std::size_t

  + ------------------------------------------------------------------------------------ +
  | QUID    | description                                                                |
  + ------- + -------------------------------------------------------------------------- +
  |   0     | response is complete (0 msgs left)                                         |
  + ------- + -------------------------------------------------------------------------- +
  | > 0     | current response ID, where ID may be translated to "msgs left"             |
  + ------------------------------------------------------------------------------------ +



# [B2/3] PAYLOAD: RES_DATA:
# -------------------------------------------------------- |

  + response data
  + serialized output SQF data structure to be
    deserialized in Arma2OA server context
